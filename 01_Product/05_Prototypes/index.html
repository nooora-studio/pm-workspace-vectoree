import { Vectoree } from "vectoree"
 
const vee = new Vectoree();
 
// pick any model. any provider. one key.
const res = await vee.chat({
  model: "x-ai/grok-4.20",
  skills: ["broker-mcp"],
  messages: [{ role: "user",
    content: "Analyze AAPL Q3 filings" }],
});
 
// charge your user for it. built in.
await vee.billing.meter(res.usage);
